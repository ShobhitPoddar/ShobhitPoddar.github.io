        const skillTagStyles = 'bg-blue-100 text-blue-800 text-sm font-medium px-3 py-1 rounded-full';

        const projectCardStyles = 'bg-white p-6 rounded-lg shadow-sm border border-slate-200 hover:shadow-lg hover:border-blue-500 transition-all duration-300';

        document.querySelectorAll('.project-card').forEach(c => c.className = projectCardStyles);


        window.onscroll = () => {

            if (window.scrollY > 300) {

                navbar.classList.remove('hidden');

            } else {

                navbar.classList.add('hidden');

            }

        };


        const observer = new IntersectionObserver((entries) => {

            entries.forEach(entry => {

                if (entry.isIntersecting) {

                    navLinks.forEach(link => {

                        link.classList.toggle('active', link.getAttribute('href').substring(1) === entry.target.id);

                    });

                }

            });

        }, { rootMargin: '-50% 0px -50% 0px' });


        sections.forEach(sec => observer.observe(sec));


        const tabs = document.querySelectorAll('.tab-btn');

        const panels = document.querySelectorAll('.tab-panel');

        const mscToggle = document.getElementById('msc-toggle');

        const mscDetails = document.getElementById('msc-details');

        const mscArrow = document.getElementById('msc-arrow');

        const tabStyles = 'inline-block py-3 px-4 text-sm font-medium text-center text-slate-500 rounded-t-lg border-b-2 border-transparent hover:text-slate-700 hover:border-slate-300 focus:outline-none transition-colors duration-300';

        ];

        const awardsData = {

            title: 'Bentley iTwin4Good 2023-2024',

            description: 'Ranked Top 12 in England & Ireland for the "Road Friction Green Storage Application" topic.'

        };

        tabs.forEach(tab => {

            tab.className = tabStyles;

            if (tab.classList.contains('tab-active')) tab.classList.add('tab-active');

                tabs.forEach(t => { t.className = tabStyles; t.classList.remove('tab-active'); });

                tab.classList.add('tab-active');

                const targetPanelId = tab.dataset.tab + '-panel';

                panels.forEach(panel => panel.classList.toggle('hidden', panel.id !== targetPanelId));

            });

        });
